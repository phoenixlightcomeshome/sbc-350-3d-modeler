    // Universal Fail-Safe Orbit Controls
    function setupUniversalOrbitControls(camera, domElement, targetVec = new THREE.Vector3(0, 1.0, 0)) {
      if (typeof THREE.OrbitControls === 'function') {
        try {
          const ctrl = new THREE.OrbitControls(camera, domElement);
          ctrl.enableDamping = true;
          ctrl.dampingFactor = 0.05;
          ctrl.target.copy(targetVec);
          return ctrl;
        } catch (e1) {}
      }
      return { update: () => {}, dispose: () => {} };
    }

        waterPump.position.set(0, 1.4, 6.8);
        waterPump.userData = { name: "High-Flow Aluminum Water Pump", desc: "Internal cast curved-vane impeller circulating 35 GPM." };
        coolingGroup.add(waterPump);

        // 2-Bolt Chrome Thermostat Housing Water Neck
        const tStatHousing = new THREE.Mesh(new THREE.CylinderGeometry(0.8, 1.0, 1.4, 16), materials.chrome);
        tStatHousing.rotation.x = 0.5;
        tStatHousing.position.set(0, 4.6, 4.6);
        tStatHousing.userData = { name: "Thermostat Housing & 180°F T-Stat", desc: "2-bolt flange neck with high-flow balanced 180-degree thermostat." };
        coolingGroup.add(tStatHousing);

        // Pulleys & Fan
        const crankPulley = new THREE.Mesh(new THREE.CylinderGeometry(2.0, 2.0, 0.8, 28), materials.chrome);
        crankPulley.rotation.x = Math.PI / 2;
        crankPulley.position.set(0, -1.0, 7.2);
        coolingGroup.add(crankPulley);

        const wpPulley = new THREE.Mesh(new THREE.CylinderGeometry(1.8, 1.8, 0.8, 28), materials.chrome);
        wpPulley.rotation.x = Math.PI / 2;
        wpPulley.position.set(0, 1.4, 7.8);
        coolingGroup.add(wpPulley);

        // Alternator & Billet Tension Bracket
        const alternator = new THREE.Mesh(new THREE.CylinderGeometry(1.6, 1.6, 2.0, 20), materials.aluminum);
        alternator.rotation.z = Math.PI / 2;
        alternator.position.set(3.8, 3.2, 6.5);
        alternator.userData = { name: "100-Amp Delco 1-Wire Alternator", desc: "Internal voltage regulator with billet tensioning turnbuckle." };
        coolingGroup.add(alternator);

        // Dual V-Belts
        const vBelt1 = new THREE.Mesh(new THREE.TorusGeometry(3.2, 0.14, 8, 32), materials.beltsRubber);
        vBelt1.position.set(0, 0.2, 7.4);
        coolingGroup.add(vBelt1);
        engineGroup.add(coolingGroup);
        parts.cooling = coolingGroup;

        // D. MECHANICAL FUEL PUMP & FUEL LINES
        const fuelGroup = new THREE.Group();
        const fuelPump = new THREE.Mesh(new THREE.CylinderGeometry(1.2, 1.2, 1.6, 16), materials.chrome);
        fuelPump.position.set(3.2, -0.8, 4.8);
        fuelPump.userData = { name: "Mechanical Fuel Pump (110 GPH)", desc: "Driven by camshaft eccentric pushrod. Regulated to 6.0 PSI." };
        fuelGroup.add(fuelPump);

        // Braided Fuel Line up to Carb Bowls
        const fuelLine = new THREE.Mesh(new THREE.CylinderGeometry(0.12, 0.12, 5.2, 12), materials.fuelHardware);
        fuelLine.position.set(2.4, 2.2, 3.6);
        fuelLine.rotation.z = -0.4;
        fuelGroup.add(fuelLine);
        engineGroup.add(fuelGroup);
        parts.fuel = fuelGroup;

        // E. HIGH-TORQUE STARTER MOTOR & SOLENOID
        const starterGroup = new THREE.Group();
        const starterMain = new THREE.Mesh(new THREE.CylinderGeometry(1.2, 1.2, 3.2, 20), materials.starterMotor);
        starterMain.rotation.x = Math.PI / 2;
        starterMain.position.set(2.6, -1.8, -4.5);
        starterGroup.add(starterMain);

        const solenoid = new THREE.Mesh(new THREE.CylinderGeometry(0.6, 0.6, 2.4, 16), materials.starterMotor);
        solenoid.rotation.x = Math.PI / 2;
        solenoid.position.set(2.6, -0.8, -4.5);
        starterGroup.add(solenoid);
        starterGroup.userData = { name: "High-Torque Mini Starter", desc: "Bolted to flywheel block pad. Pinion backlash shimmed to 0.035\"." };
        engineGroup.add(starterGroup);
        parts.starter = starterGroup;

        scene.add(engineGroup);
        renderer.render(scene, camera);

        // Raycasting for Every Mechanic Part
        const raycaster = new THREE.Raycaster();
        const mouse = new THREE.Vector2();
        const onCanvasClick = (e) => {
          const rect = renderer.domElement.getBoundingClientRect();
          mouse.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
          mouse.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
          raycaster.setFromCamera(mouse, camera);
          const hits = raycaster.intersectObjects(engineGroup.children, true);
          if (hits.length > 0) {
            let obj = hits[0].object;
            while (obj.parent && obj.parent !== engineGroup) {
              if (obj.userData && obj.userData.name) break;
              obj = obj.parent;
            }
            if (obj.userData && obj.userData.name) {
              setSelectedPart(obj.userData);
            }
          }
        };
        renderer.domElement.addEventListener('click', onCanvasClick);

        // Action Loop
        let animId;
        let lastTime = performance.now();
        const animate = (now) => {
          animId = requestAnimationFrame(animate);
          const dt = Math.min((now - lastTime) / 1000, 0.1);
          lastTime = now;

          if (controls.update) controls.update();

          const f = explodedAmountRef.current;
          if (parts.heads) parts.heads.children[0].position.set(-3.8 - f * 4.5, 3.2 + f * 4.5, 0);
          if (parts.heads) parts.heads.children[1].position.set(3.8 + f * 4.5, 3.2 + f * 4.5, 0);
          if (parts.ignition) parts.ignition.position.y = f * 6.0;
          if (parts.cooling) parts.cooling.position.z = f * 5.0;
          if (parts.lube) parts.lube.position.y = -f * 4.5;
          if (parts.starter) parts.starter.position.x = 2.6 + f * 4.0;

          if (isRunningRef.current) {
            const radPerSec = (rpmRef.current / 60) * 2 * Math.PI;
            if (parts.crank) parts.crank.rotation.z += radPerSec * dt;
            if (parts.cam) parts.cam.rotation.z += (radPerSec * dt) / 2;

            pistonNodes.forEach((node) => {
              const theta = (parts.crank ? parts.crank.rotation.z : 0) + node.angleOffset;
              node.group.position.y = node.baseY + Math.sin(theta) * 0.45;
            });
          }

          renderer.render(scene, camera);
        };
        animate(performance.now());

        return () => {
          cancelAnimationFrame(animId);
          if (renderer.domElement) renderer.domElement.removeEventListener('click', onCanvasClick);
          if (controls.dispose) controls.dispose();
          renderer.dispose();
        };
      }, [activeTab, viewMode, cutawayActive, mechanicIsolation]);

      return (
        <div className="w-full min-h-screen flex flex-col bg-chevyDark text-gray-200">
          
          {/* TOP HEADER */}
          <header ref={headerRef} className="flex-shrink-0 bg-chevyPanel border-b border-gray-700 px-3 py-2 flex flex-col gap-2 z-20 shadow-lg">
            <div className="flex flex-wrap items-center justify-between gap-2">
              <div className="flex items-center space-x-2.5">
                <div className="w-8 h-8 rounded bg-chevyOrange flex items-center justify-center font-black text-sm text-white shadow-md">
                  350
                </div>
                <div>
                  <h1 className="font-bold text-sm tracking-wide text-white flex items-center gap-1.5">
                    {activeEngine.shortName} <span className="text-[10px] px-1.5 py-0.2 rounded bg-chevyOrange/20 text-chevyOrange border border-chevyOrange/40 font-mono">ALL MECHANIC COMPONENTS</span>
                  </h1>
                  <p className="text-[10px] text-gray-400 font-mono">
                    Ray's Baseline: 5.700" Rods | 9.025" Deck | 64cc Heads | .039" Gasket | .015" Deck Clr | Quench .054"
                  </p>
                </div>
              </div>

              {/* NAV TABS */}
              <nav className="flex space-x-1 overflow-x-auto max-w-full">
                {[
                  { id: 'block3d', label: '1. 3D Engine & All Components', icon: '⚙️' },
                  { id: 'specs', label: '2. Mechanic Specs & Tolerances', icon: '📐' },
                  { id: 'shop', label: '3. Parts Counter & BOM', icon: '🏬' }
                ].map((tab) => (
                  <button
                    key={tab.id}
                    onClick={() => setActiveTab(tab.id)}
                    className={`px-2.5 py-1 rounded text-xs font-semibold whitespace-nowrap transition-all flex items-center gap-1 ${
                      activeTab === tab.id
                        ? 'bg-chevyOrange text-white shadow'
                        : 'text-gray-400 hover:text-white hover:bg-gray-800'
                    }`}
                  >
                    <span>{tab.icon}</span>
                    <span>{tab.label}</span>
                  </button>
                ))}
              </nav>
            </div>
          </header>

          {/* MAIN WORKSPACE */}
          <main className="flex-1 w-full relative">
            
            {/* TAB 1: 3D ENGINE & COMPLETE MECHANIC'S HARDWARE */}
            {activeTab === 'block3d' && (
              <div 
                style={{
                  position: 'relative',
                  width: '100%',
                  height: `calc(100vh - ${headerHeight}px)`,
                  minHeight: '480px'
                }}
                className="overflow-hidden blueprint-grid"
              >
                <div ref={canvasContainerRef} style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%', touchAction: 'none' }} className="cursor-grab active:cursor-grabbing" />

                {/* MECHANIC'S SUB-SYSTEM ISOLATION BAR (TOP-LEFT) */}
                <div className="absolute top-3 left-3 z-10 flex flex-col gap-2 max-w-[calc(100%-24px)]">
                  <div className="flex flex-wrap items-center gap-1.5 bg-chevyPanel/95 backdrop-blur border border-chevyOrange/60 p-1.5 rounded-lg shadow-xl text-xs font-mono">
                    <span className="text-[10px] font-bold text-chevyGold uppercase px-1">🔧 ISOLATE HARDWARE:</span>
                    {[
                      { id: 'all', label: 'Full Engine' },
                      { id: 'rotating', label: 'Rotating (5.700" Rods/Crank)' },
                      { id: 'valvetrain', label: 'Valvetrain (Cam/Lifters/Valves)' },
                      { id: 'ignition', label: 'Ignition (HEI/Plugs/Wires)' },
                      { id: 'lubrication', label: 'Oil Pump & Pickup' },
                      { id: 'cooling', label: 'Water Pump & Belts' },
                      { id: 'fuel', label: 'Fuel Pump & Lines' },
                      { id: 'starter', label: 'Mini Starter' }
                    ].map(btn => (
                      <button
                        key={btn.id}
                        onClick={() => setMechanicIsolation(btn.id)}
                        className={`px-2 py-0.5 rounded ${mechanicIsolation === btn.id ? 'bg-chevyOrange text-white font-bold' : 'text-gray-400 hover:text-white'}`}
                      >
                        {btn.label}
                      </button>
                    ))}
                  </div>

                  {/* Cutaway Toggle & Action Button */}
                  <div className="flex flex-wrap items-center gap-1.5">
                    <button
                      onClick={() => setCutawayActive(!cutawayActive)}
                      className={`px-2.5 py-1 text-xs rounded font-bold border shadow ${
                        cutawayActive ? 'bg-cadCyan border-cadCyan text-black' : 'bg-chevyPanel border-gray-700 text-gray-300'
                      }`}
                    >
                      {cutawayActive ? '🔪 CUTAWAY ACTIVE' : '🔍 OPEN CUTAWAY SLICE'}
                    </button>

                    <button
                      onClick={() => setIsRunning(!isRunning)}
                      className={`px-3 py-1 text-xs font-bold rounded shadow-xl border flex items-center gap-1.5 ${
                        isRunning ? 'bg-emerald-600 border-emerald-400 text-white animate-pulse' : 'bg-chevyPanel border-gray-700 text-gray-300'
                      }`}
                    >
                      <span>{isRunning ? '⏹ STOP KINEMATICS' : '▶ RUN CYCLE ACTION'}</span>
                    </button>
                  </div>
                </div>

                {/* Exploded Disassembly Slider */}
                <div className="absolute bottom-4 left-4 right-4 sm:right-auto sm:w-96 z-10 bg-chevyPanel/95 backdrop-blur border border-gray-700 rounded-xl p-3 shadow-2xl space-y-2">
                  <div className="flex items-center justify-between border-b border-gray-700 pb-1.5">
                    <span className="text-[11px] font-bold tracking-wider text-gray-300">BLOWN-APART DISASSEMBLY</span>
                    <span className="text-xs font-mono text-chevyOrange font-bold">{Math.round(explodedAmount * 100)}%</span>
                  </div>
                  <input
                    type="range" min="0" max="1" step="0.01"
                    value={explodedAmount}
                    onChange={(e) => setExplodedAmount(parseFloat(e.target.value))}
                    className="w-full accent-chevyOrange cursor-pointer h-2"
                  />
                  <div className="text-[10px] text-gray-400 font-mono">
                    Tap any mechanic part (Distributor, Starter, Water Pump, Oil Pickup) to inspect real specs.
                  </div>
                </div>

                {/* Mechanic Component Inspector HUD */}
                {selectedPart && (
                  <div className="absolute top-16 right-3 left-3 sm:left-auto sm:w-84 z-20 bg-chevyPanel/95 backdrop-blur border border-chevyOrange/60 rounded-xl p-4 shadow-2xl font-mono text-xs">
                    <div className="flex items-start justify-between">
                      <span className="text-[10px] uppercase tracking-wider text-chevyOrange font-bold">MECHANIC INSPECTION</span>
                      <button onClick={() => setSelectedPart(null)} className="text-gray-400 hover:text-white font-bold p-1">✕</button>
                    </div>
                    <h3 className="font-bold text-sm text-white mt-1">{selectedPart.name}</h3>
                    <p className="text-xs text-gray-300 mt-1 leading-relaxed font-sans">{selectedPart.desc}</p>
                  </div>
                )}
              </div>
            )}

            {/* TAB 2: MECHANIC SPECS & TOLERANCES */}
            {activeTab === 'specs' && (
              <div className="p-4 sm:p-6 max-w-6xl mx-auto w-full space-y-5">
                <div className="flex justify-between items-center">
                  <div>
                    <h2 className="text-lg font-bold text-white">Real-World Mechanic Operating Specs & Clearances</h2>
                    <p className="text-xs text-gray-400">Ray's Confirmed SBC 350 Baseline Standards</p>
                  </div>
                  <div className="text-xs font-mono bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 px-3 py-1.5 rounded">
                    QUENCH DISTANCE: 0.054" (.039" gasket + .015" deck) - PASS
                  </div>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-3">
                  {activeEngine.mechanicSpecs.map((s, i) => (
                    <div key={i} className="p-3 bg-chevyPanel border border-gray-800 rounded-xl space-y-1 text-xs font-mono">
                      <div className="text-gray-400 font-bold">{s.label}</div>
                      <div className="text-sm font-bold text-chevyGold">{s.value}</div>
                      <p className="text-[10px] text-gray-500">{s.notes}</p>
                    </div>
                  ))}
                </div>

                <div className="overflow-x-auto bg-chevyPanel rounded-xl border border-gray-800">
                  <table className="w-full text-left text-xs font-mono">
                    <thead className="bg-chevyDark text-gray-400">
                      <tr>
                        <th className="p-2.5">FASTENER LOCATION</th>
                        <th className="p-2.5">THREAD SIZE</th>
                        <th className="p-2.5">TORQUE SPEC</th>
                        <th className="p-2.5">LUBRICATION & CRITICAL NOTES</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-gray-800">
                      {activeEngine.torqueSpecs.map((t, idx) => (
                        <tr key={idx}>
                          <td className="p-2.5 text-white font-bold">{t.item}</td>
                          <td className="p-2.5">{t.size}</td>
                          <td className="p-2.5 text-chevyOrange font-bold">{t.stock} / {t.arp}</td>
                          <td className="p-2.5 text-gray-400">{t.notes}</td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>
            )}

            {/* TAB 3: PARTS COUNTER & BOM */}
            {activeTab === 'shop' && (
              <div className="p-4 sm:p-6 max-w-6xl mx-auto w-full space-y-5">
                <h2 className="text-lg font-bold text-white">Auto Parts Counter SKU Cross-Reference & Pricing</h2>
                <div className="bg-chevyPanel border border-gray-800 rounded-xl p-4">
                  <table className="w-full text-left text-xs font-mono">
                    <thead className="bg-chevyDark text-gray-400">
                      <tr>
                        <th className="p-2">CATEGORY</th>
                        <th className="p-2">COMPONENT</th>
                        <th className="p-2">MANUFACTURER PART NUMBER</th>
                        <th className="p-2 text-right">EST. PRICE</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-gray-800">
                      {activeEngine.bom.map((b, i) => (
                        <tr key={i}>
                          <td className="p-2 font-bold text-white">{b.cat}</td>
                          <td className="p-2 text-gray-300">{b.part}</td>
                          <td className="p-2 text-chevyGold">{b.num}</td>
                          <td className="p-2 text-right font-bold text-emerald-400">${b.price.toFixed(2)}</td>
                        </tr>
                      ))}
                      <tr className="bg-chevyDark font-bold">
                        <td colSpan="3" className="p-2.5 text-right text-white">TOTAL ESTIMATED PARTS COST:</td>
                        <td className="p-2.5 text-right text-chevyGold font-black">
                          ${activeEngine.bom.reduce((s, p) => s + p.price, 0).toFixed(2)}
                        </td>
                      </tr>
                    </tbody>
                  </table>
                </div>
              </div>
            )}

          </main>
        </div>
      );
    }

    ReactDOM.createRoot(document.getElementById('root')).render(<SBC350App />);
  </script>
</body>
