    // 4. COMPATIBILITY & CROSS-FITMENT MATCHING ENGINE
    // =========================================================================
    function auditTransmissionCompatibility(engine, trans, vehicleInfo) {
      const ePattern = engine.pattern;
      const tPattern = trans.nativePattern;

      let isDirectBoltOn = false;
      let adapterRequired = "None (Direct Factory Bolt-On)";
      let hardwareNotes = "";
      let bellhousing = "";

      // DIRECT BOLT-ON CHECKS
      if (ePattern === tPattern || (tPattern === "GM_Dual_BOP_Chevy" && ePattern === "GM_Chevy_V8")) {
        isDirectBoltOn = true;
        bellhousing = "Direct OEM Bellhousing Pattern Match";
        if (ePattern === "GM_Chevy_V8") {
          hardwareNotes = "Uses standard Chevy 6-bolt bellhousing. Match starter nose to 153 vs 168-tooth flywheel.";
        } else if (ePattern === "Ford_SBF_6Bolt") {
          hardwareNotes = "Uses Ford small block 6-bolt bellhousing. Match 28 oz vs 50 oz flywheel unbalance.";
        }
      } 
      // CROSS-PLATFORM SWAP ADAPTER CHECKS
      else {
        isDirectBoltOn = false;
        if (ePattern === "GM_Chevy_V8" && trans.id === "nissan_cd009") {
          adapterRequired = "Collins / Maverick SBC-to-CD009 Billet Adapter Plate";
          hardwareNotes = "Uses custom stepped flywheel with Chevy bolt pattern and 350Z clutch disc.";
          bellhousing = "Requires machined front bellhousing bell cut";
        } else if (ePattern === "GM_Chevy_V8" && (trans.id === "tremecT56" || trans.id === "tremec_tkx")) {
          adapterRequired = "QuickTime RM-6023 SFI Steel Bellhousing (Chevy V8 to Tremec T56 Face)";
          hardwareNotes = "Direct bolt-in with 26-spline 11\" clutch disc and hydraulic throwout bearing.";
          bellhousing = "SFI Spec 6.1 Explosive Proof Bellhousing";
        } else if (ePattern === "GM_Chevy_V8" && trans.id === "zf_8hp70_universal") {
          adapterRequired = "DomTube / Sound Performance 8HP-to-SBC Adapter Plate + Crank Hub Adapter";
          hardwareNotes = "Requires standalone HTG Tuning or Turbolamik TCU transmission controller.";
          bellhousing = "Billet 6061-T6 Adapter Plate";
        } else if (ePattern === "GM_LS_Modular" && trans.nativePattern === "GM_Chevy_V8") {
          adapterRequired = "GM 0.400\" Crankshaft Flange Spacer (GM Part #12563532)";
          hardwareNotes = "Adapts early short-crank LS to traditional TH350/TH400/Powerglide torque converter hub.";
          bellhousing = "Standard GM bellhousing with top-center bolt omitted";
        } else {
          adapterRequired = "Aftermarket Multi-Pattern Billet Bellhousing / Adapter Plate";
          hardwareNotes = "Custom flywheel spacing and pilot bushing extension required.";
          bellhousing = "Custom SFI Adapter Bellhousing";
        }
      }

      return {
        isDirectBoltOn,
        adapterRequired,
        hardwareNotes,
        bellhousing,
        tunnelClearance: vehicleInfo?.tunnelSize || "Verify floorpan transmission crossmember clearance"
      };
    }

    // Universal Fail-Safe Orbit Controls
    function setupUniversalOrbitControls(camera, domElement, targetVec = new THREE.Vector3(0, 1.0, 0)) {
      if (typeof THREE.OrbitControls === 'function') {
        try {
          const ctrl = new THREE.OrbitControls(camera, domElement);
          ctrl.enableDamping = true;
          ctrl.dampingFactor = 0.05;
          ctrl.target.copy(targetVec);
          return ctrl;
        } catch (e1) {}
      }
      return { update: () => {}, dispose: () => {} };
    }

                  <div ref={transCanvasContainerRef} style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%', touchAction: 'none' }} className="cursor-grab active:cursor-grabbing" />

                  {/* Transmission Category Filter Bar */}
                  <div className="absolute top-3 left-3 z-10 flex flex-wrap gap-1.5 max-w-[calc(100%-24px)] text-xs font-mono">
                    <span className="text-[10px] font-bold text-chevyGold uppercase px-1 py-1">FAMILY:</span>
                    {['ALL', 'GM Classic Automatic', 'GM Electronic Overdrive', 'Classic Muscle Manual', 'Modern Overdrive Manual', 'JDM & Exotic Swap'].map(cat => (
                      <button
                        key={cat}
                        onClick={() => setTransCategoryFilter(cat)}
                        className={`px-2 py-0.5 rounded text-[11px] ${transCategoryFilter === cat ? 'bg-chevyOrange text-white font-bold' : 'bg-chevyPanel/90 text-gray-400 hover:text-white'}`}
                      >
                        {cat}
                      </button>
                    ))}
                  </div>

                  {/* Primary Transmission Selector */}
                  <div className="absolute top-12 left-3 z-10 flex flex-wrap items-center gap-2">
                    <select
                      value={selectedTrans}
                      onChange={(e) => {
                        setSelectedTrans(e.target.value);
                        setStallSpeed(TRANSMISSIONS_DB[e.target.value].defaultStall || 2200);
                      }}
                      className="bg-chevyPanel/95 border-2 border-chevyOrange rounded px-2.5 py-1 text-xs font-bold text-white shadow-2xl"
                    >
                      {filteredTransmissions.map((t) => (
                        <option key={t.id} value={t.id}>[{t.type}] {t.name}</option>
                      ))}
                    </select>

                    <button
                      onClick={() => setTransRunning(!transRunning)}
                      className={`px-3 py-1 text-xs font-bold rounded shadow-xl border ${
                        transRunning ? 'bg-emerald-600 border-emerald-400 text-white animate-pulse' : 'bg-chevyPanel border-gray-700 text-gray-300'
                      }`}
                    >
                      {transRunning ? '⏹ STOP GEAR ROTATION' : '▶ SPIN INPUT / OUTPUT SHAFTS'}
                    </button>
                  </div>

                  {/* Blown-Apart Disassembly Slider */}
                  <div className="absolute bottom-4 left-4 right-4 sm:right-auto sm:w-96 z-10 bg-chevyPanel/95 backdrop-blur border border-gray-700 rounded-xl p-3 shadow-2xl space-y-2">
                    <div className="flex items-center justify-between border-b border-gray-700 pb-1.5">
                      <span className="text-[11px] font-bold tracking-wider text-gray-300">BLOWN-APART DISASSEMBLY</span>
                      <span className="text-xs font-mono text-chevyGold font-bold">{Math.round(transExploded * 100)}%</span>
                    </div>
                    <input
                      type="range" min="0" max="1" step="0.01"
                      value={transExploded}
                      onChange={(e) => setTransExploded(parseFloat(e.target.value))}
                      className="w-full accent-chevyGold cursor-pointer h-2"
                    />
                    <div className="text-[10px] text-gray-400 font-mono">
                      Separates Bellhousing, Torque Converter / Clutch, Main Case, and Tailshaft.
                    </div>
                  </div>
                </div>

                {/* COMPATIBILITY & CROSS-FITMENT REPORT PANEL */}
                <div className="w-full lg:w-96 p-4 bg-chevyDark border-l border-gray-800 flex flex-col gap-4 overflow-y-auto">
                  <div className="pb-2 border-b border-gray-700">
                    <h3 className="text-xs font-bold text-white uppercase tracking-wider font-mono">DRIVETRAIN COMPATIBILITY REPORT</h3>
                    <p className="text-[11px] text-gray-400 mt-0.5">{activeEngine.shortName} ➔ {currentTrans.name}</p>
                  </div>

                  {/* Direct Bolt-On Status Badge */}
                  <div className={`p-3 rounded-xl border flex items-center gap-2.5 font-mono text-xs ${
                    compatibilityReport.isDirectBoltOn
                      ? 'bg-emerald-500/10 border-emerald-500/40 text-emerald-300'
                      : 'bg-amber-500/10 border-amber-500/40 text-amber-300'
                  }`}>
                    <span className="text-lg">{compatibilityReport.isDirectBoltOn ? '✓' : '⚙️'}</span>
                    <div>
                      <div className="font-bold">{compatibilityReport.isDirectBoltOn ? 'DIRECT OEM BOLT-ON' : 'SWAP ADAPTER REQUIRED'}</div>
                      <div className="text-[10px] text-gray-400">{compatibilityReport.bellhousing}</div>
                    </div>
                  </div>

                  {/* Bellhousing, Flywheel & Starter Notes */}
                  <div className="bg-chevyPanel p-3 rounded-xl border border-gray-800 space-y-2 text-xs font-mono">
                    <span className="text-chevyGold font-bold text-[10px] uppercase block">HARDWARE & BELLHOUSING MATING:</span>
                    <p className="text-gray-300 text-[11px] leading-relaxed">{compatibilityReport.hardwareNotes}</p>
                    <div className="pt-2 border-t border-gray-700/80 text-[10px] text-gray-400">
                      <div>Engine Flywheel Pattern: <span className="text-white font-bold">{activeEngine.flywheelTeeth}</span></div>
                      <div>Balance Requirement: <span className="text-white font-bold">{activeEngine.balance}</span></div>
                      <div>Adapter Kit: <span className="text-cadCyan font-bold">{compatibilityReport.adapterRequired}</span></div>
                    </div>
                  </div>

                  {/* Vehicle Chassis Tunnel Fitment */}
                  <div className="bg-chevyPanel p-3 rounded-xl border border-gray-800 space-y-2 text-xs font-mono">
                    <span className="text-cadCyan font-bold text-[10px] uppercase block">FLOORPAN TUNNEL FITMENT ({selectedModel}):</span>
                    <p className="text-emerald-400 text-[11px] leading-relaxed font-bold">{compatibilityReport.tunnelClearance}</p>
                    <div className="text-[10px] text-gray-400">
                      Case Profile: {currentTrans.tunnelProfile}
                    </div>
                  </div>

                  {/* Road Speed & Gear Reduction Table */}
                  <div className="bg-chevyPanel p-3 rounded-xl border border-gray-800 space-y-2 text-xs font-mono">
                    <div className="flex justify-between items-center">
                      <span className="text-white font-bold text-[10px] uppercase">SPEED IN GEAR (REAR AXLE {rearGear}:1)</span>
                      <select value={rearGear} onChange={(e) => setRearGear(parseFloat(e.target.value))} className="bg-black/60 rounded px-1.5 py-0.5 text-[10px] text-chevyGold">
                        <option value={3.08}>3.08:1</option>
                        <option value={3.42}>3.42:1</option>
                        <option value={3.73}>3.73:1</option>
                        <option value={4.10}>4.10:1</option>
                        <option value={4.56}>4.56:1</option>
                      </select>
                    </div>

                    <div className="overflow-x-auto">
                      <table className="w-full text-left text-[11px]">
                        <thead className="text-gray-400 border-b border-gray-700">
                          <tr>
                            <th className="py-1">GEAR</th>
                            <th className="py-1">RATIO</th>
                            <th className="py-1 text-emerald-400">5500 RPM</th>
                            <th className="py-1 text-gray-400">@ 65 MPH</th>
                          </tr>
                        </thead>
                        <tbody className="divide-y divide-gray-800">
                          {speedInGearTable.map(row => (
                            <tr key={row.gear}>
                              <td className="py-1 font-bold text-white">Gear {row.gear}</td>
                              <td className="py-1 text-chevyGold">{row.ratio.toFixed(2)}</td>
                              <td className="py-1 text-emerald-400">{row.mphAt5500} MPH</td>
                              <td className="py-1 text-gray-300">{row.cruiseRpmAt65 > 5500 ? 'N/A' : `${row.cruiseRpmAt65} RPM`}</td>
                            </tr>
                          ))}
                        </tbody>
                      </table>
                    </div>
                  </div>
                </div>
              </div>
            )}

            {/* TAB 2: 3D ENGINE BLOCK */}
            {activeTab === 'block3d' && (
              <div style={{ position: 'relative', width: '100%', height: `calc(100vh - ${headerHeight}px)`, minHeight: '480px' }} className="overflow-hidden blueprint-grid">
                <div ref={canvasContainerRef} style={{ position: 'absolute', top: 0, left: 0, width: '100%', height: '100%', touchAction: 'none' }} className="cursor-grab active:cursor-grabbing" />
              </div>
            )}

            {/* TAB 3: SCHEMATICS & FASTENERS */}
            {activeTab === 'schematics' && (
              <div className="p-4 sm:p-6 max-w-6xl mx-auto w-full space-y-5">
                <h2 className="text-lg font-bold text-white">Fasteners & 17-Bolt Cylinder Head Torquing Sequence</h2>
                <div className="p-4 bg-chevyPanel rounded-xl border border-gray-800 flex flex-col gap-3">
                  <span className="text-xs font-bold text-chevyGold">17-Bolt Sequence (Stage 1: 25 ft-lbs ➔ Stage 2: 45 ft-lbs ➔ Stage 3: 65 ft-lbs)</span>
                  <div className="p-3 bg-chevyDark rounded-lg flex justify-between items-center max-w-lg mx-auto w-full">
                    {[13, 9, 5, 1, 3, 7, 11, 15, 17].map(n => (
                      <div key={n} className="w-7 h-7 rounded-full bg-chevyOrange flex items-center justify-center font-bold text-xs text-white">{n}</div>
                    ))}
                  </div>
                </div>
              </div>
            )}

            {/* TAB 4: PARTS COUNTER & BOM */}
            {activeTab === 'shop' && (
              <div className="p-4 sm:p-6 max-w-6xl mx-auto w-full space-y-5">
                <h2 className="text-lg font-bold text-white">Auto Parts Counter & Cross-Fitment Catalog</h2>
                <div className="text-xs font-mono bg-emerald-500/10 border border-emerald-500/30 text-emerald-400 px-3 py-1.5 rounded inline-block">
                  RAY'S CONFIRMED QUENCH AUDIT: {activeEngine.quenchDistance}" (.039" gasket + .015" deck) - PASS
                </div>
              </div>
            )}

          </main>
        </div>
      );
    }
