export type TraceEngine = 'vtracer' | 'opencv' | 'manual_contour';

export type TargetManufacturingProcess =
  | 'cnc_5_axis'
  | 'lpbf_slm_dmls'
  | 'ded_am'
  | 'superplastic_forming'
  | 'near_net_forging';

export type ManufacturabilityStatus = 'PASS' | 'WARN_ESTIMATE' | 'FLAG_INFEASIBLE';

export interface Point2D {
  x: number;
  y: number;
}

export interface CalibratedScalePlane {
  planeId: string;
  referencePixels: number;
  referenceInches: number;
  inchesPerPixel: number;
  mmPerInch: number;
  calibrationNote: string;
  needsPhysicalVerification: boolean;
  sourceRef: string;
}

export interface TracedContourInput {
  contourId: string;
  partName: string;
  traceEngine: TraceEngine;
  /** Contour vertices in pixel coordinates on the calibrated scale plane */
  pixelPoints: Point2D[];
  /** Extrusion / feature depth in inches */
  extrudeDepthIn: number;
  /** Draft angle in degrees (0 if none specified) */
  draftAngleDeg?: number;
  /** Minimum wall or feature thickness in inches if known; if omitted, derived from bounding contour */
  minFeatureSizeIn?: number;
  /** Whether part/feature involves an internal/solid-state join rather than an extruded contour */
  isInternalOrSolidStateJoin?: boolean;
  /** Optional density in g/cm^3 from a verified/curated source (e.g. 2.70 or 4.43); never invented */
  densityGCm3?: number;
  sourceRef: string;
}

export interface ProcessManufacturabilityRule {
  process: TargetManufacturingProcess;
  label: string;
  /** Minimum wall/feature size in inches; null when no single published limit exists in chat sources */
  minFeatureSizeIn: number | null;
  /** Maximum recommended extrusion/feature depth in inches; null when unverified in chat sources */
  maxExtrudeDepthIn: number | null;
  /** Minimum draft angle in degrees if required/flagged; null when unverified in chat sources */
  minDraftAngleDeg: number | null;
  /** Whether internal/solid-state join features cannot be represented as a simple extruded contour */
  rejectsExtrudedContour: boolean;
  ruleSummary: string;
  sourceRef: string;
}

export interface ManufacturabilityFinding {
  code:
    | 'BELOW_MIN_FEATURE_SIZE'
    | 'UNVERIFIED_FEATURE_SIZE'
    | 'EXCEEDS_MAX_DEPTH'
    | 'UNVERIFIED_DEPTH'
    | 'INSUFFICIENT_DRAFT'
    | 'UNVERIFIED_DRAFT'
    | 'INCOMPATIBLE_CONTOUR_TOPOLOGY'
    | 'VENDOR_OR_ESTIMATE_LIMIT'
    | 'NEEDS_RAY_MEASUREMENT';
  severity: 'INFO' | 'WARN' | 'BLOCK';
  message: string;
  sourceRef: string;
}

export interface EvaluatedPartGeometry {
  contourId: string;
  partName: string;
  process: TargetManufacturingProcess;
  status: ManufacturabilityStatus;
  scalePlane: CalibratedScalePlane;
  /** Converted vertices in inches */
  pointsIn: Point2D[];
  /** Converted vertices in millimeters */
  pointsMm: Point2D[];
  widthIn: number;
  heightIn: number;
  extrudeDepthIn: number;
  extrudeDepthMm: number;
  draftAngleDeg: number;
  /** 2D area in square inches (from planar polygon shoelace formula) */
  areaIn2: number;
  /** Extruded volume in cubic inches (`areaIn2 * extrudeDepthIn`) */
  volumeIn3: number;
  /** Extruded volume in cubic centimeters (`volumeIn3 * 16.387064`) */
  volumeCm3: number;
  /** Mass in grams only when `densityGCm3` is supplied; otherwise null */
  estimatedMassGrams: number | null;
  findings: ManufacturabilityFinding[];
  browserPreviewSpec: {
    renderer: 'three.js (MIT) ExtrudeGeometry + STLExporter';
    clientSideOnly: boolean;
    note: string;
  };
  serverStepExportSpec: {
    engine: 'build123d (Apache-2.0) on OpenCASCADE / OCCT (LGPL-2.1)';
    format: 'STEP AP242 (ISO 10303-242 MBD)';
    endpointHint: string;
    note: string;
  };
}

function computePolygonAreaIn2(points: Point2D[]): number {
  if (points.length < 3) return 0;
  let twiceArea = 0;
  for (let i = 0; i < points.length; i++) {
    const p1 = points[i];
    const p2 = points[(i + 1) % points.length];
    twiceArea += p1.x * p2.y - p2.x * p1.y;
  }
  return Math.abs(twiceArea) * 0.5;
}

function estimateMinimumSpanIn(pointsIn: Point2D[]): number {
  if (pointsIn.length < 2) return 0;
  let minSpan = Infinity;
  for (let i = 0; i < pointsIn.length; i++) {
    const p1 = pointsIn[i];
    const p2 = pointsIn[(i + 1) % pointsIn.length];
    const dist = Math.hypot(p2.x - p1.x, p2.y - p1.y);
    if (dist > 0 && dist < minSpan) {
      minSpan = dist;
    }
  }
  return Number.isFinite(minSpan) ? minSpan : 0;
}

/**
 * Evaluates a traced 2D contour + extrusion depth against the calibrated scale plane
 * and target manufacturing process rules, flagging infeasible or unverified features
 * instead of silently fabricating specs.
 */
export function evaluateTracedPart(
  input: TracedContourInput,
  process: TargetManufacturingProcess,
  scalePlane: CalibratedScalePlane = CALIBRATED_TRUE_SCALE_PLANE
): EvaluatedPartGeometry {
  const rule = PROCESS_CAPABILITY_RULES[process];
  const pointsIn: Point2D[] = input.pixelPoints.map((pt) => ({
    x: Number((pt.x * scalePlane.inchesPerPixel).toFixed(5)),
    y: Number((pt.y * scalePlane.inchesPerPixel).toFixed(5))
  }));
  const pointsMm: Point2D[] = pointsIn.map((pt) => ({
    x: Number((pt.x * scalePlane.mmPerInch).toFixed(4)),
    y: Number((pt.y * scalePlane.mmPerInch).toFixed(4))
  }));

  const xs = pointsIn.map((p) => p.x);
  const ys = pointsIn.map((p) => p.y);
  const widthIn = xs.length > 0 ? Number((Math.max(...xs) - Math.min(...xs)).toFixed(5)) : 0;
  const heightIn = ys.length > 0 ? Number((Math.max(...ys) - Math.min(...ys)).toFixed(5)) : 0;

  const extrudeDepthIn = Number(input.extrudeDepthIn.toFixed(5));
  const extrudeDepthMm = Number((extrudeDepthIn * scalePlane.mmPerInch).toFixed(4));
  const draftAngleDeg = input.draftAngleDeg ?? 0;

  const areaIn2 = Number(computePolygonAreaIn2(pointsIn).toFixed(6));
  const volumeIn3 = Number((areaIn2 * extrudeDepthIn).toFixed(6));
  const volumeCm3 = Number((volumeIn3 * 16.387064).toFixed(6));

  const estimatedMassGrams =
    typeof input.densityGCm3 === 'number' && Number.isFinite(input.densityGCm3)
      ? Number((volumeCm3 * input.densityGCm3).toFixed(4))
      : null;

  const findings: ManufacturabilityFinding[] = [
    {
      code: 'NEEDS_RAY_MEASUREMENT',
      severity: 'WARN',
      message: `Scale plane (${scalePlane.referencePixels} px = ${scalePlane.referenceInches.toFixed(3)} in) and final part dimensions/tolerances require physical verification [NEEDS RAY'S MEASUREMENT].`,
      sourceRef: scalePlane.sourceRef
    }
  ];

  let hasBlock = false;
  let hasWarn = false;

  if (input.pixelPoints.length < 3 || areaIn2 <= 0) {
    hasBlock = true;
    findings.push({
      code: 'INCOMPATIBLE_CONTOUR_TOPOLOGY',
      severity: 'BLOCK',
      message: 'Contour has fewer than 3 non-collinear points or zero enclosed area; cannot extrude a valid closed part.',
      sourceRef: input.sourceRef
    });
  }

  if (input.isInternalOrSolidStateJoin || rule.rejectsExtrudedContour) {
    hasBlock = true;
    findings.push({
      code: 'INCOMPATIBLE_CONTOUR_TOPOLOGY',
      severity: 'BLOCK',
      message: `Feature/process (${rule.label}) is not valid as a standalone 2D extruded contour; flag for dedicated solid/process modeling.`,
      sourceRef: rule.sourceRef
    });
  }

  if (extrudeDepthIn <= 0) {
    hasBlock = true;
    findings.push({
      code: 'EXCEEDS_MAX_DEPTH',
      severity: 'BLOCK',
      message: `Extrusion depth (${extrudeDepthIn.toFixed(4)} in) must be greater than zero to form a 3D part.`,
      sourceRef: input.sourceRef
    });
  }

  const effectiveFeatureIn =
    typeof input.minFeatureSizeIn === 'number'
      ? input.minFeatureSizeIn
      : estimateMinimumSpanIn(pointsIn);

  if (typeof input.minFeatureSizeIn !== 'number') {
    hasWarn = true;
    findings.push({
      code: 'UNVERIFIED_FEATURE_SIZE',
      severity: 'WARN',
      message: `Explicit minFeatureSizeIn not provided; using smallest contour segment (${effectiveFeatureIn.toFixed(4)} in) as proxy — verify actual wall/boss thickness.`,
      sourceRef: input.sourceRef
    });
  }

  if (rule.minFeatureSizeIn !== null && effectiveFeatureIn > 0 && effectiveFeatureIn < rule.minFeatureSizeIn) {
    hasBlock = true;
    findings.push({
      code: 'BELOW_MIN_FEATURE_SIZE',
      severity: 'BLOCK',
      message: `Feature size (${effectiveFeatureIn.toFixed(4)} in / ${(effectiveFeatureIn * 25.4).toFixed(2)} mm) is below ${rule.label} reference threshold (${rule.minFeatureSizeIn.toFixed(4)} in / ${(rule.minFeatureSizeIn * 25.4).toFixed(2)} mm).`,
      sourceRef: rule.sourceRef
    });
  }

  if (rule.minDraftAngleDeg !== null && draftAngleDeg < rule.minDraftAngleDeg) {
    hasWarn = true;
    findings.push({
      code: 'INSUFFICIENT_DRAFT',
      severity: 'WARN',
      message: `Draft angle (${draftAngleDeg}°) is below ${rule.label} reference minimum (${rule.minDraftAngleDeg}°) or lacks explicit machining stock allowance.`,
      sourceRef: rule.sourceRef
    });
  }

  if (rule.maxExtrudeDepthIn !== null && extrudeDepthIn > rule.maxExtrudeDepthIn) {
    hasBlock = true;
    findings.push({
      code: 'EXCEEDS_MAX_DEPTH',
      severity: 'BLOCK',
      message: `Extrusion depth (${extrudeDepthIn.toFixed(4)} in) exceeds ${rule.label} limit (${rule.maxExtrudeDepthIn.toFixed(4)} in).`,
      sourceRef: rule.sourceRef
    });
  } else if (rule.maxExtrudeDepthIn === null) {
    hasWarn = true;
    findings.push({
      code: 'UNVERIFIED_DEPTH',
      severity: 'WARN',
      message: `No universal max depth published for ${rule.label} in chat sources; check aspect ratio, tool reach, overhang/support, or die limits [ESTIMATE].`,
      sourceRef: rule.sourceRef
    });
  }

  if (rule.minDraftAngleDeg === null && input.draftAngleDeg === undefined) {
    hasWarn = true;
    findings.push({
      code: 'UNVERIFIED_DRAFT',
      severity: 'WARN',
      message: `No draft angle specified and no universal draft rule published for ${rule.label}; verify process/tooling requirements [ESTIMATE].`,
      sourceRef: rule.sourceRef
    });
  }

  findings.push({
    code: 'VENDOR_OR_ESTIMATE_LIMIT',
    severity: 'INFO',
    message: rule.ruleSummary,
    sourceRef: rule.sourceRef
  });

  const status: ManufacturabilityStatus = hasBlock
    ? 'FLAG_INFEASIBLE'
    : hasWarn
      ? 'WARN_ESTIMATE'
      : 'PASS';

  return {
    contourId: input.contourId,
    partName: input.partName,
    process,
    status,
    scalePlane,
    pointsIn,
    pointsMm,
    widthIn,
    heightIn,
    extrudeDepthIn,
    extrudeDepthMm,
    draftAngleDeg,
    areaIn2,
    volumeIn3,
    volumeCm3,
    estimatedMassGrams,
    findings,
    browserPreviewSpec: {
      renderer: 'three.js (MIT) ExtrudeGeometry + STLExporter',
      clientSideOnly: true,
      note: 'Client-side preview and mesh export only (after vtracer MIT/WASM or OpenCV.js Apache-2.0 contour trace). Use server build123d/OpenCASCADE for real STEP AP242 solids.'
    },
    serverStepExportSpec: {
      engine: 'build123d (Apache-2.0) on OpenCASCADE / OCCT (LGPL-2.1)',
      format: 'STEP AP242 (ISO 10303-242 MBD)',
      endpointHint: '/api/extrude-step',
      note: 'Send calibrated inch/mm points and depth to server build123d/OCCT script to generate a true B-rep STEP AP242 file.'
    }
  };
}

/**
 * Generates an ASCII/text STL (in inches) from the extruded top/bottom/side triangles
 * when `allowFlaggedExport` is true; blocks silently exporting infeasible parts unless explicitly overridden.
 */
export function exportPartToStlInches(
  evaluated: EvaluatedPartGeometry,
  options: { allowFlaggedExport?: boolean } = {}
): { ok: boolean; reason?: string; content?: string } {
  if (evaluated.status === 'FLAG_INFEASIBLE' && !options.allowFlaggedExport) {
    return {
      ok: false,
      reason: `Export blocked: part ${evaluated.partName} is flagged INFEASIBLE for ${evaluated.process}. Resolve findings or pass allowFlaggedExport: true.`
    };
  }

  const pts = evaluated.pointsIn;
  if (pts.length < 3 || evaluated.extrudeDepthIn <= 0) {
    return { ok: false, reason: 'Invalid polygon or non-positive extrusion depth in inches.' };
  }

  const z0 = 0;
  const z1 = evaluated.extrudeDepthIn;
  const triangles: Array<[Point2D & { z: number }, Point2D & { z: number }, Point2D & { z: number }]> = [];

  // Fan triangulation across XY contour for bottom and top caps
  for (let i = 1; i < pts.length - 1; i++) {
    triangles.push([
      { x: pts[0].x, y: pts[0].y, z: z0 },
      { x: pts[i + 1].x, y: pts[i + 1].y, z: z0 },
      { x: pts[i].x, y: pts[i].y, z: z0 }
    ]);
    triangles.push([
      { x: pts[0].x, y: pts[0].y, z: z1 },
      { x: pts[i].x, y: pts[i].y, z: z1 },
      { x: pts[i + 1].x, y: pts[i + 1].y, z: z1 }
    ]);
  }

  // Side quads split into two triangles per edge
  for (let i = 0; i < pts.length; i++) {
    const a = pts[i];
    const b = pts[(i + 1) % pts.length];
    const b0 = { x: a.x, y: a.y, z: z0 };
    const b1 = { x: b.x, y: b.y, z: z0 };
    const t1 = { x: b.x, y: b.y, z: z1 };
    const t0 = { x: a.x, y: a.y, z: z1 };
    triangles.push([b0, b1, t1]);
    triangles.push([b0, t1, t0]);
  }

  const lines: string[] = [`solid ${evaluated.contourId}_inches`];
  for (const [v1, v2, v3] of triangles) {
    const ux = v2.x - v1.x;
    const uy = v2.y - v1.y;
    const uz = v2.z - v1.z;
    const vx = v3.x - v1.x;
    const vy = v3.y - v1.y;
    const vz = v3.z - v1.z;
    let nx = uy * vz - uz * vy;
    let ny = uz * vx - ux * vz;
    let nz = ux * vy - uy * vx;
    const len = Math.hypot(nx, ny, nz) || 1;
    nx /= len;
    ny /= len;
    nz /= len;

    lines.push(`  facet normal ${nx.toExponential(6)} ${ny.toExponential(6)} ${nz.toExponential(6)}`);
    lines.push('    outer loop');
    lines.push(`      vertex ${v1.x.toExponential(6)} ${v1.y.toExponential(6)} ${v1.z.toExponential(6)}`);
    lines.push(`      vertex ${v2.x.toExponential(6)} ${v2.y.toExponential(6)} ${v2.z.toExponential(6)}`);
    lines.push(`      vertex ${v3.x.toExponential(6)} ${v3.y.toExponential(6)} ${v3.z.toExponential(6)}`);
    lines.push('    endloop');
    lines.push('  endfacet');
  }
  lines.push(`endsolid ${evaluated.contourId}_inches`);

  return { ok: true, content: lines.join('\n') };
}

/**
 * Exports the 2D calibrated contour (and optional extrusion depth note) as an inch-unit DXF R12-style LWPOLYLINE/LINE exchange text.
 */
export function exportContourToDxfInches(
  evaluated: EvaluatedPartGeometry,
  options: { allowFlaggedExport?: boolean } = {}
): { ok: boolean; reason?: string; content?: string } {
  if (evaluated.status === 'FLAG_INFEASIBLE' && !options.allowFlaggedExport) {
    return {
      ok: false,
      reason: `DXF export blocked: part ${evaluated.partName} is flagged INFEASIBLE for ${evaluated.process}.`
    };
  }

  const pts = evaluated.pointsIn;
  if (pts.length < 2) {
    return { ok: false, reason: 'At least 2 points are required for DXF export.' };
  }

  // $MEASUREMENT = 0 (Imperial / Inches)
  const dxf: string[] = [
    '0', 'SECTION',
    '2', 'HEADER',
    '9', '$MEASUREMENT',
    '70', '0',
    '0', 'ENDSEC',
    '0', 'SECTION',
    '2', 'ENTITIES'
  ];

  for (let i = 0; i < pts.length; i++) {
    const p1 = pts[i];
    const p2 = pts[(i + 1) % pts.length];
    dxf.push(
      '0', 'LINE',
      '8', `CONTOUR_IN_${evaluated.process.toUpperCase()}`,
      '10', p1.x.toFixed(6),
      '20', p1.y.toFixed(6),
      '30', '0.000000',
      '11', p2.x.toFixed(6),
      '21', p2.y.toFixed(6),
      '31', '0.000000'
    );
  }

  dxf.push('0', 'ENDSEC', '0', 'EOF');
  return { ok: true, content: dxf.join('\n') };
}

/**
 * Exports the extruded part as a Wavefront OBJ mesh in inches (1 unit = 1 inch).
 */
export function exportPartToObjInches(
  evaluated: EvaluatedPartGeometry,
  options: { allowFlaggedExport?: boolean } = {}
): { ok: boolean; reason?: string; content?: string } {
  if (evaluated.status === 'FLAG_INFEASIBLE' && !options.allowFlaggedExport) {
    return {
      ok: false,
      reason: `OBJ export blocked: part ${evaluated.partName} is flagged INFEASIBLE for ${evaluated.process}.`
    };
  }

  const pts = evaluated.pointsIn;
  if (pts.length < 3 || evaluated.extrudeDepthIn <= 0) {
    return { ok: false, reason: 'Invalid polygon or non-positive extrusion depth in inches.' };
  }

  const z0 = 0;
  const z1 = evaluated.extrudeDepthIn;
  const n = pts.length;
  const lines: string[] = [
    `# Unit: inches (1 OBJ unit = 1 in); calibrated scale: ${evaluated.scalePlane.referencePixels} px = ${evaluated.scalePlane.referenceInches.toFixed(3)} in`,
    `# Process: ${evaluated.process} | Status: ${evaluated.status}`,
    `o ${evaluated.contourId}_in`
  ];

  // Bottom vertices 1..n, top vertices (n+1)..(2n)
  for (const p of pts) {
    lines.push(`v ${p.x.toFixed(6)} ${p.y.toFixed(6)} ${z0.toFixed(6)}`);
  }
  for (const p of pts) {
    lines.push(`v ${p.x.toFixed(6)} ${p.y.toFixed(6)} ${z1.toFixed(6)}`);
  }

  // Bottom cap
  const botIndices = Array.from({ length: n }, (_, idx) => n - idx).join(' ');
  lines.push(`f ${botIndices}`);

  // Top cap
  const topIndices = Array.from({ length: n }, (_, idx) => n + 1 + idx).join(' ');
  lines.push(`f ${topIndices}`);

  // Side faces
  for (let i = 0; i < n; i++) {
    const next = (i + 1) % n;
    const b1 = i + 1;
    const b2 = next + 1;
    const t2 = n + next + 1;
    const t1 = n + i + 1;
    lines.push(`f ${b1} ${b2} ${t2} ${t1}`);
  }

  return { ok: true, content: lines.join('\n') };
}

/**
 * Derives Young's modulus E and (when absent) Poisson's ratio nu from bulk (K) and shear (G) moduli.
 * Formulas:
 *   nu = (3K - 2G) / (2(3K + G))
 *   E  = 9KG / (3K + G)
 * Note: Materials Project summary elasticity fields contain VRH bulk/shear moduli (such as k_vrh / g_vrh);
 * k_vrh is bulk modulus, NOT Young's modulus. Exact schema field names should be confirmed against
 * https://docs.materialsproject.org/ when integrating.
 */
export function deriveElasticityFromBulkAndShear(
  bulkModulusGpa: number | null,
  shearModulusGpa: number | null,
  suppliedPoissonsRatio: number | null
): DerivedElasticity {
  const kValid = isFinitePositiveNumber(bulkModulusGpa);
  const gValid = isFinitePositiveNumber(shearModulusGpa);

  if (!kValid || !gValid) {
    return {
      bulkModulusGpa: kValid ? bulkModulusGpa : null,
      shearModulusGpa: gValid ? shearModulusGpa : null,
      poissonsRatio: isFiniteNumber(suppliedPoissonsRatio) ? suppliedPoissonsRatio : null,
      poissonsRatioDerived: false,
      youngsModulusGpa: null,
      youngsModulusDerived: false,
      derivationNote:
        'Young\'s modulus not derived: valid positive bulk (K) and shear (G) moduli were not both available in response [ESTIMATE screening only; verify against https://docs.materialsproject.org/].'
    };
  }

  const denom = 3 * bulkModulusGpa + shearModulusGpa;
  const derivedNu = (3 * bulkModulusGpa - 2 * shearModulusGpa) / (2 * denom);
  const hasSuppliedNu = isFiniteNumber(suppliedPoissonsRatio);
  const effectiveNu = hasSuppliedNu ? suppliedPoissonsRatio : derivedNu;
  const youngsModulusGpa = (9 * bulkModulusGpa * shearModulusGpa) / denom;

  return {
    bulkModulusGpa,
    shearModulusGpa,
    poissonsRatio: isFiniteNumber(effectiveNu) ? Number(effectiveNu.toFixed(6)) : null,
    poissonsRatioDerived: !hasSuppliedNu && isFiniteNumber(derivedNu),
    youngsModulusGpa: Number(youngsModulusGpa.toFixed(4)),
    youngsModulusDerived: true,
    derivationNote: hasSuppliedNu
      ? 'Young\'s modulus E derived from K and G via E = 9KG / (3K + G) using supplied Poisson\'s ratio [ESTIMATE screening only; k_vrh is bulk modulus, not Young\'s modulus].'
      : 'Young\'s modulus E = 9KG / (3K + G) and Poisson\'s ratio nu = (3K - 2G) / (2(3K + G)) derived from K and G [ESTIMATE screening only; verify exact schema on https://docs.materialsproject.org/].'
  };
}

}

  if (!raw || typeof raw !== 'object') return null;

  // Defensive extraction across possible elasticity sub-object shapes; confirm exact schema in MP docs.

export class MaterialsProjectFetcher {
  private readonly apiKeyEnvVar: string;
  private readonly endpointUrl: string;
  private readonly batchSize: number;
  private readonly cacheTtlMs: number;
  private readonly cacheFilePath: string | undefined;
  private readonly fetchFn: typeof fetch;
  private readonly memoryCache = new Map<string, CacheEntry>();

  constructor(options: MaterialsProjectFetcherOptions = {}) {
    assertServerOnlyRuntime();

    this.apiKeyEnvVar = options.apiKeyEnvVar ?? DEFAULT_ENV_VAR;
    this.endpointUrl = options.endpointUrl ?? DEFAULT_ENDPOINT;
    this.batchSize = Math.max(1, options.batchSize ?? DEFAULT_BATCH_SIZE);
    this.cacheTtlMs = Math.max(0, options.cacheTtlMs ?? DEFAULT_TTL_MS);
    this.cacheFilePath = options.cacheFilePath;
    this.fetchFn = options.fetchFn ?? fetch;

    if (this.cacheFilePath) {
      this.loadDiskCacheIntoMemory();
    }
  }

  private getVaultApiKey(): string {
    assertServerOnlyRuntime();
    const envMap = typeof process !== 'undefined' ? process.env : undefined;
    const key = envMap ? envMap[this.apiKeyEnvVar] : undefined;
    if (!key || key.trim().length === 0) {
      throw new VaultKeyMissingError(this.apiKeyEnvVar);
    }
    return key.trim();
  }

  private loadDiskCacheIntoMemory(): void {
    if (!this.cacheFilePath || !existsSync(this.cacheFilePath)) return;
    try {
      const raw = readFileSync(this.cacheFilePath, 'utf8');
      const parsed = JSON.parse(raw) as Record<string, CacheEntry>;
      for (const [matId, entry] of Object.entries(parsed)) {
        if (entry && entry.record && typeof entry.cachedAtMs === 'number') {
          this.memoryCache.set(matId, entry);
        }
      }
    } catch {
      // Ignore unreadable/corrupt cache file and rebuild cleanly on subsequent writes
    }
  }

  private persistMemoryCacheToDisk(): void {
    if (!this.cacheFilePath) return;
    const dir = dirname(this.cacheFilePath);
    if (dir && dir !== '.' && !existsSync(dir)) {
      mkdirSync(dir, { recursive: true });
    }
    const obj: Record<string, CacheEntry> = {};
    for (const [matId, entry] of this.memoryCache.entries()) {
      obj[matId] = entry;
    }
    writeFileSync(this.cacheFilePath, JSON.stringify(obj, null, 2), 'utf8');
  }

  private isFresh(entry: CacheEntry, nowMs: number): boolean {
    return nowMs - entry.cachedAtMs <= this.cacheTtlMs;
  }

  private chunkIds(ids: string[]): string[][] {
    const chunks: string[][] = [];
    for (let i = 0; i < ids.length; i += this.batchSize) {
      chunks.push(ids.slice(i, i + this.batchSize));
    }
    return chunks;
  }

  private async fetchBatchPost(batchIds: string[], apiKey: string): Promise<ScreenedMaterialRecord[]> {
    let response: Response;
    try {
      // Batches material_ids in one POST request; verify exact POST body schema against https://docs.materialsproject.org/
      response = await this.fetchFn(this.endpointUrl, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Accept: 'application/json',
          'X-API-KEY': apiKey
        },
        body: JSON.stringify({
          material_ids: batchIds,
          fields: ['material_id', 'formula_pretty', 'density', 'elasticity']
        })
      });
    } catch (err) {
      const detail = err instanceof Error ? err.message : String(err);
      throw new MaterialsProjectNetworkError(detail);
    }

    if (response.status === 401 || response.status === 403) {
      throw new MaterialsProjectAuthError(
        response.status,
        `Materials Project authentication/authorization failed (HTTP ${response.status}). Verify the key stored in "${this.apiKeyEnvVar}" in the Secure Vault.`
      );
    }

    if (response.status === 429) {
      const retryAfterSec = parseRetryAfterSeconds(response.headers.get('Retry-After'));
      throw new MaterialsProjectRateLimitError(
        retryAfterSec,
        'Materials Project API rate limit exceeded.'
      );
    }

    if (!response.ok) {
      throw new MaterialsProjectNetworkError(
        `Unexpected HTTP ${response.status} (${response.statusText}) from Materials Project API.`
      );
    }

    let payload: unknown;
    try {
      payload = await response.json();
    } catch (err) {
      const detail = err instanceof Error ? err.message : String(err);
      throw new MaterialsProjectNetworkError(`Invalid JSON in Materials Project response: ${detail}`);
    }

    const docs =
      payload && typeof payload === 'object' && Array.isArray((payload as { data?: unknown }).data)
        ? (payload as { data: unknown[] }).data
        : [];

    const retrievedAt = new Date().toISOString();
    const normalized: ScreenedMaterialRecord[] = [];
    for (const doc of docs) {
      const rec = normalizeSummaryDoc(doc, retrievedAt);
      if (rec) normalized.push(rec);
    }
    return normalized;
  }

  /**
   * Retrieves screening records for requested material IDs using cache-first logic,
   * batching missing/expired IDs into POST /materials/summary requests.
   * On HTTP 429, serves cached entries (even if stale) whenever available; otherwise throws
   * MaterialsProjectRateLimitError with Retry-After guidance.
   */
  async getMaterials(materialIds: string[]): Promise<FetchMaterialsResult> {
    assertServerOnlyRuntime();

    const uniqueIds = Array.from(
      new Set(materialIds.map((id) => id.trim()).filter((id) => id.length > 0))
    );
    if (uniqueIds.length === 0) {
      return {
        records: [],
        servedFromCache: false,
        staleCacheUsed: false,
        warnings: []
      };
    }

    const nowMs = Date.now();
    const cachedRecords = new Map<string, ScreenedMaterialRecord>();
    const idsToFetch: string[] = [];
    let staleCacheUsed = false;

    for (const id of uniqueIds) {
      const existing = this.memoryCache.get(id);
      if (existing && this.isFresh(existing, nowMs)) {
        cachedRecords.set(id, existing.record);
      } else {
        idsToFetch.push(id);
      }
    }

    const warnings: string[] = [];

    if (idsToFetch.length > 0) {
      const apiKey = this.getVaultApiKey();
      const batches = this.chunkIds(idsToFetch);

      for (const batch of batches) {
        try {
          const fetched = await this.fetchBatchPost(batch, apiKey);
          const fetchTimeMs = Date.now();
          for (const rec of fetched) {
            this.memoryCache.set(rec.material_id, { record: rec, cachedAtMs: fetchTimeMs });
            cachedRecords.set(rec.material_id, rec);
          }
          this.persistMemoryCacheToDisk();
        } catch (err) {
          if (err instanceof MaterialsProjectRateLimitError) {
            const fallbackAvailable = batch.some((id) => this.memoryCache.has(id));
            if (fallbackAvailable) {
              staleCacheUsed = true;
              warnings.push(`${err.message} Serving cached entries for available IDs.`);
              for (const id of batch) {
                const fallbackEntry = this.memoryCache.get(id);
                if (fallbackEntry && !cachedRecords.has(id)) {
                  cachedRecords.set(id, fallbackEntry.record);
                }
              }
              continue;
            }
            throw err;
          }
          throw err;
        }
      }
    }

    const orderedRecords: ScreenedMaterialRecord[] = [];
    for (const id of uniqueIds) {
      const rec = cachedRecords.get(id);
      if (rec) orderedRecords.push(rec);
    }

    return {
      records: orderedRecords,
      servedFromCache: idsToFetch.length === 0 || staleCacheUsed,
      staleCacheUsed,
      warnings
    };
  }

      });
    } else if (obj.baseTq !== BASELINE_CONSTANTS.expectedCarbTq) {

      });
    }
  }

  if (obj.mpfiHp !== undefined && obj.mpfiHp !== BASELINE_CONSTANTS.expectedMpfiHp) {
    issues.push({
      table,
      field: 'mpfiHp',
      expected: String(BASELINE_CONSTANTS.expectedMpfiHp),

export function computeDisplacementCid(
  boreIn: number,
  strokeIn: number,
  cylinders: number
): number {
  const raw = (Math.PI / 4) * boreIn * boreIn * strokeIn * cylinders;
  return Number(raw.toFixed(2));
}

export function validateEngineData(
  engineData: EngineDataTableInput | undefined,
  toleranceCid: number = BASELINE_CONSTANTS.defaultDisplacementToleranceCid
): {
  issues: ValidationIssue[];
  displacement: DisplacementCheckResult;
  baseTqDiscrepancies: BaseTqDiscrepancyRecord[];
} {
  const table = engineData?.tableName ?? 'engineData';
  const issues: ValidationIssue[] = [];
  const baseTqDiscrepancies: BaseTqDiscrepancyRecord[] = [];

  if (!engineData) {
    issues.push({
      table,
      field: 'engineData',
      expected: 'Present engineData table',
      actual: 'Missing',
      severity: 'FAIL',
      message: 'engineData table is missing; no default values invented.'
    });
    return {
      issues,
      displacement: {
        boreIn: null,
        strokeIn: null,
        cylinders: null,
        recomputedCid: null,
        reportedCid: null,
        toleranceCid,
        withinTolerance: false
      },
      baseTqDiscrepancies
    };
  }

  if (!hasNonEmptyCitation(engineData)) {
    issues.push({
      table,
      field: 'sourceRef',
      expected: 'Non-empty table source or sourceRef',
      actual: 'Missing',
      severity: 'FAIL',
      message: `${table} is missing a top-level source/sourceRef citation.`
    });
  }

  const boreIn = typeof engineData.boreIn === 'number' ? engineData.boreIn : null;
  const strokeIn = typeof engineData.strokeIn === 'number' ? engineData.strokeIn : null;
  const cylinders = typeof engineData.cylinders === 'number' ? engineData.cylinders : null;
  const reportedCid =
    typeof engineData.displacementCid === 'number' ? engineData.displacementCid : null;

  let recomputedCid: number | null = null;
  let withinTolerance = false;

  if (boreIn === null || strokeIn === null || cylinders === null) {
    issues.push({
      table,
      field: 'displacementInputs',
      expected: `boreIn (${BASELINE_CONSTANTS.expectedBoreIn}), strokeIn (${BASELINE_CONSTANTS.expectedStrokeIn}), cylinders (${BASELINE_CONSTANTS.expectedCylinders})`,
      actual: `boreIn=${boreIn}, strokeIn=${strokeIn}, cylinders=${cylinders}`,
      severity: 'FAIL',
      message: `${table} is missing bore, stroke, or cylinder count required to recompute CID.`
    });
  } else {
    recomputedCid = computeDisplacementCid(boreIn, strokeIn, cylinders);
    if (reportedCid === null) {
      issues.push({
        table,
        field: 'displacementCid',
        expected: `${recomputedCid.toFixed(2)} CID`,
        actual: 'Missing',
        severity: 'FAIL',
        message: `${table} is missing reported displacementCid (recomputed: ${recomputedCid.toFixed(2)} CID).`
      });
    } else {
      const diff = Math.abs(recomputedCid - reportedCid);
      withinTolerance = diff <= toleranceCid;
      if (!withinTolerance) {
        issues.push({
          table,
          field: 'displacementCid',
          expected: `${recomputedCid.toFixed(2)} CID (±${toleranceCid} CID tolerance; reference ${BASELINE_CONSTANTS.expectedDisplacementCid.toFixed(2)} CID)`,
          actual: `${reportedCid.toFixed(2)} CID (diff ${diff.toFixed(4)} CID)`,
          severity: 'FAIL',
          message: `${table} displacementCid ${reportedCid} differs from recomputed ${recomputedCid.toFixed(2)} CID beyond ${toleranceCid} CID tolerance.`
        });
      }
    }
  }

  checkBaselineHpTqAndBaseTq(table, engineData, issues, baseTqDiscrepancies);
  checkRecordCitationsAndFlags(table, engineData.records, 'record', issues);

  return {
    issues,
    displacement: {
      boreIn,
      strokeIn,
      cylinders,
      recomputedCid,
      reportedCid,
      toleranceCid,
      withinTolerance
    },
    baseTqDiscrepancies
  };
}

export function validateAll(tables: ValidationTablesInput): ValidationReport {
  const toleranceCid =
    typeof tables.displacementToleranceCid === 'number' &&
    Number.isFinite(tables.displacementToleranceCid) &&
    tables.displacementToleranceCid >= 0
      ? tables.displacementToleranceCid
      : BASELINE_CONSTANTS.defaultDisplacementToleranceCid;

  const issues: ValidationIssue[] = [];
  const baseTqDiscrepancies: BaseTqDiscrepancyRecord[] = [];

  const engineResult = validateEngineData(tables.engineData, toleranceCid);
  issues.push(...engineResult.issues);
  baseTqDiscrepancies.push(...engineResult.baseTqDiscrepancies);

  validateManufacturingMatrix(tables.manufacturingMatrix, issues, baseTqDiscrepancies);
  validateTechDrawingsCatalog(tables.techDrawingsCatalog, issues, baseTqDiscrepancies);
  validateAerospaceMatrix(tables.aerospaceMatrix, issues);

  const failureCount = issues.filter((i) => i.severity === 'FAIL').length;
  const discrepancyCount = issues.filter((i) => i.severity === 'DISCREPANCY').length;

  return {
    passed: failureCount === 0 && discrepancyCount === 0,
    totalIssues: issues.length,
    failureCount,
    discrepancyCount,
    displacement: engineResult.displacement,
    baseTqDiscrepancies,
    issues
  };
}

export function formatValidationReport(report: ValidationReport): string {
  const lines: string[] = [
    `Status: ${report.passed ? 'PASS' : 'FAIL / ACTION REQUIRED'}`,
    `Summary: totalIssues=${report.totalIssues}, failures=${report.failureCount}, discrepancies=${report.discrepancyCount}`,
    `Displacement: recomputed=${report.displacement.recomputedCid ?? 'N/A'} CID, reported=${report.displacement.reportedCid ?? 'N/A'} CID, tolerance=±${report.displacement.toleranceCid} CID, withinTolerance=${report.displacement.withinTolerance}`
  ];

  if (report.baseTqDiscrepancies.length > 0) {
    lines.push('BaseTq Discrepancies (Ray ruling required — values not altered):');
    for (const d of report.baseTqDiscrepancies) {
      lines.push(
        `  - [${d.table}.${d.field}] found=${d.valueFound} vs reference=${d.referenceValue} — ${d.note}`
      );
    }
  }

  if (report.issues.length > 0) {
    lines.push('Issues:');
    for (const item of report.issues) {
      const rulingTag = item.requiresRayRuling ? ' [RAY RULING REQUIRED]' : '';
      lines.push(
        `  - [${item.severity}]${rulingTag} ${item.table}.${item.field}: expected="${item.expected}", actual="${item.actual}" — ${item.message}`
      );
    }
  } else {
    lines.push('Issues: none');
  }

  return lines.join('\n');
}
